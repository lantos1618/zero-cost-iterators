import Vw.Plan
/-! `groupFold(key, init, step)`: a terminal that keeps ONE accumulator per distinct key (first-seen
key order) and folds each key's items in original order. Arbitrary item type `α`, key type `κ`
(decidable equality, e.g. `String`), accumulator type `β`; `step` is any pure total function
(no associativity). Upstream map/filter fuse into the same single traversal via `View.foldPath`. -/
namespace Vw
variable {α β κ : Type} [DecidableEq κ]

/-- implementation step: walk the key table; on a hit, step that key's accumulator in place; on a
miss, append a new entry `(key x, step init x)` at the END (first-seen order) -/
def gIns (key : α → κ) (init : β) (step : β → α → β) : List (κ × β) → α → List (κ × β)
  | [], x => [(key x, step init x)]
  | (k, b) :: t, x => if k = key x then (k, step b x) :: t else (k, b) :: gIns key init step t x

/-- independent reference, stable group-then-fold: keys = the `distinct` reference of the key
sequence (first occurrence order); each group = the stable subsequence with that key, folded left -/
def groupRef (key : α → κ) (init : β) (step : β → α → β) (xs : List α) : List (κ × β) :=
  (dedupRef (xs.map key)).map fun k => (k, (xs.filter (fun y => key y = k)).foldl step init)

/-! ### facts about the `distinct` reference -/
theorem mem_dedupRef : ∀ (n : Nat) (l : List κ), l.length ≤ n → ∀ k, k ∈ dedupRef l ↔ k ∈ l
  | _, [], _, k => by simp [dedupRef_nil]
  | 0, _ :: _, h, _ => by simp at h
  | n + 1, y :: ys, h, k => by
    rw [dedupRef_cons]
    have ih := mem_dedupRef n (ys.filter (fun z => z ≠ y))
      (Nat.le_trans (List.length_filter_le _ _) (by simp at h; omega)) k
    by_cases hk : k = y <;> simp_all

theorem nodup_dedupRef : ∀ (n : Nat) (l : List κ), l.length ≤ n → (dedupRef l).Nodup
  | _, [], _ => by simp [dedupRef_nil]
  | 0, _ :: _, h => by simp at h
  | n + 1, y :: ys, h => by
    rw [dedupRef_cons]
    have hl : (ys.filter (fun z => z ≠ y)).length ≤ n :=
      Nat.le_trans (List.length_filter_le _ _) (by simp at h; omega)
    refine List.nodup_cons.2 ⟨fun hm => ?_, nodup_dedupRef n _ hl⟩
    have := (mem_dedupRef n _ hl y).1 hm
    simp at this

theorem dedupRef_snoc : ∀ (n : Nat) (l : List κ), l.length ≤ n → ∀ k,
    dedupRef (l ++ [k]) = if k ∈ l then dedupRef l else dedupRef l ++ [k]
  | _, [], _, k => by simp [dedupRef_cons, dedupRef_nil]
  | 0, _ :: _, h, _ => by simp at h
  | n + 1, y :: ys, h, k => by
    have hl : (ys.filter (fun z => z ≠ y)).length ≤ n :=
      Nat.le_trans (List.length_filter_le _ _) (by simp at h; omega)
    rw [List.cons_append, dedupRef_cons, dedupRef_cons, List.filter_append]
    by_cases hk : k = y
    · subst hk; simp
    · have e : List.filter (fun z => decide (z ≠ y)) [k] = [k] := by simp [hk]
      rw [e, dedupRef_snoc n _ hl k]
      by_cases hm : k ∈ ys <;> simp [hk, hm]

/-! ### one implementation step = the reference on the extended prefix -/
theorem gIns_hit (key : α → κ) (init : β) (step : β → α → β) (x : α) (F G : κ → β) :
    ∀ K : List κ, K.Nodup → key x ∈ K → G (key x) = step (F (key x)) x → (∀ k, k ≠ key x → G k = F k) →
    gIns key init step (K.map fun k => (k, F k)) x = K.map fun k => (k, G k)
  | [], _, hm, _, _ => by simp at hm
  | k :: K, hn, hm, hG, hF => by
    simp only [List.map_cons, gIns]
    by_cases hk : k = key x
    · subst hk
      have hnot : key x ∉ K := (List.nodup_cons.1 hn).1
      simp only [if_true, hG, List.cons.injEq, true_and]
      refine List.map_congr_left fun k' hk' => ?_
      rw [hF k' (fun e => hnot (e ▸ hk'))]
    · simp only [hk, if_false, hF k hk, List.cons.injEq, true_and]
      exact gIns_hit key init step x F G K (List.nodup_cons.1 hn).2 (by simp at hm; exact hm.resolve_left (fun e => hk e.symm)) hG hF

theorem gIns_miss (key : α → κ) (init : β) (step : β → α → β) (x : α) (F : κ → β) :
    ∀ K : List κ, key x ∉ K →
    gIns key init step (K.map fun k => (k, F k)) x = (K.map fun k => (k, F k)) ++ [(key x, step init x)]
  | [], _ => rfl
  | k :: K, hm => by
    have hk : k ≠ key x := fun e => hm (by simp [e])
    simp only [List.map_cons, gIns, hk, if_false, List.cons_append]
    rw [gIns_miss key init step x F K (fun h => hm (by simp [h]))]

theorem gIns_ref (key : α → κ) (init : β) (step : β → α → β) (l : List α) (x : α) :
    gIns key init step (groupRef key init step l) x = groupRef key init step (l ++ [x]) := by
  unfold groupRef
  rw [List.map_append, List.map_singleton, dedupRef_snoc _ _ (Nat.le_refl _)]
  have hG : ∀ k, ((l ++ [x]).filter (fun y => key y = k)).foldl step init =
      if key x = k then step ((l.filter (fun y => key y = k)).foldl step init) x
      else (l.filter (fun y => key y = k)).foldl step init := by
    intro k; by_cases h : key x = k <;> simp [List.filter_append, h]
  by_cases hm : key x ∈ l.map key
  · simp only [hm, if_true]
    rw [gIns_hit key init step x (fun k => (l.filter (fun y => key y = k)).foldl step init)
      (fun k => ((l ++ [x]).filter (fun y => key y = k)).foldl step init) _ (nodup_dedupRef _ _ (Nat.le_refl _))
      ((mem_dedupRef _ _ (Nat.le_refl _) _).2 hm) (by simp only []; rw [hG]; simp) (fun k hk => by simp only []; rw [hG]; simp [Ne.symm hk])]
  · simp only [hm, if_false]
    rw [gIns_miss key init step x (fun k => (l.filter (fun y => key y = k)).foldl step init) _ (fun h => hm ((mem_dedupRef _ _ (Nat.le_refl _) _).1 h)),
      List.map_append, List.map_singleton]
    have hnil : l.filter (fun y => key y = key x) = [] := by
      rw [List.filter_eq_nil_iff]; intro y hy e; exact hm (List.mem_map.2 ⟨y, hy, by simpa using e⟩)
    congr 1
    · refine List.map_congr_left fun k hk => ?_
      have hne : key x ≠ k := fun e => hm ((mem_dedupRef _ _ (Nat.le_refl _) _).1 (e ▸ hk))
      rw [hG]; simp [hne]
    · simp [List.filter_append, hnil]

/-- groupFold implementation (a left fold of `gIns` from the empty table) = stable group-then-fold -/
theorem groupFold_correct (key : α → κ) (init : β) (step : β → α → β) (xs : List α) :
    xs.foldl (gIns key init step) [] = groupRef key init step xs := by
  suffices h : ∀ pre, xs.foldl (gIns key init step) (groupRef key init step pre) =
      groupRef key init step (pre ++ xs) by simpa [groupRef, dedupRef_nil] using h []
  induction xs with
  | nil => simp
  | cons x xs ih => intro pre; simp only [List.foldl_cons, gIns_ref, ih]; simp

/-- retained state: one entry per distinct key seen, nothing per item -/
theorem groupFold_state (key : α → κ) (init : β) (step : β → α → β) (xs : List α) :
    (xs.foldl (gIns key init step) []).map Prod.fst = dedupRef (xs.map key) := by
  rw [groupFold_correct]; simp [groupRef, Function.comp_def]

end Vw

namespace Vw.View
variable {α β κ : Type} [DecidableEq κ]

/-- terminal `groupFold(key, init, step)`. `.parallelize` is REFUSED: per-key state is an ordered
fold and no lawful chunk summary (associative merge with identity) is supplied. -/
def groupFold (v : View α) (key : α → κ) (init : β) (step : β → α → β) : Except String (List (κ × β)) :=
  if v.chunks.isSome then .error (refusedPar "groupFold: ordered per-key fold, no lawful chunk summary")
  else .ok (v.foldPath [] (gIns key init step))

theorem groupFold_ok (v : View α) (key : α → κ) (init : β) (step : β → α → β) (h : v.chunks = none) :
    v.groupFold key init step = .ok (groupRef key init step (den v.plan)) := by
  simp [groupFold, h, foldPath_eq, groupFold_correct]

theorem groupFold_refused (v : View α) (key : α → κ) (init : β) (step : β → α → β) (k : Nat) :
    ∃ e, (v.parallelize k).groupFold key init step = .error e := by
  simp [groupFold, parallelize]

/-- the fused filter → map → groupFold call is literally ONE left fold over the source whose body
tests, transforms and inserts: no list between adapters -/
theorem groupFold_fused_loop {σ : Type} (xs : List σ) (p : σ → Bool) (f : σ → α)
    (key : α → κ) (init : β) (step : β → α → β) :
    (xs.view.filter p |>.map f |>.fuse).groupFold key init step =
      .ok (xs.foldl (fun t x => if p x then gIns key init step t (f x) else t) []) := rfl

/-- and the same for foldl -/
theorem foldl_fused_loop {σ : Type} (xs : List σ) (p : σ → Bool) (f : σ → α) (seed : β) (step : β → α → β) :
    (xs.view.filter p |>.map f |>.fuse).foldl seed step =
      .ok (xs.foldl (fun b x => if p x then step b (f x) else b) seed) := rfl

/-- generic semantic theorem at the call-site shape, arbitrary types, no associativity -/
theorem foldl_view_generic {σ : Type} (xs : List σ) (p : σ → Bool) (f : σ → α) (seed : β) (step : β → α → β) :
    (xs.view.filter p |>.map f |>.fuse).foldl seed step = .ok (((xs.filter p).map f).foldl step seed) := by
  rw [foldl_ok _ _ _ rfl]; rfl

theorem groupFold_view_generic {σ : Type} (xs : List σ) (p : σ → Bool) (f : σ → α)
    (key : α → κ) (init : β) (step : β → α → β) :
    (xs.view.filter p |>.map f |>.fuse).groupFold key init step = .ok (groupRef key init step ((xs.filter p).map f)) := by
  rw [groupFold_ok _ _ _ _ rfl]; rfl
end Vw.View
